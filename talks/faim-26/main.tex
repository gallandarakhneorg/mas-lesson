\sidecite{Wooldridge09,CossentinoGaudHilaireGallandKoukam2010_1,rodriguez_sarl:_2014}
\begin{frame}[t]{{Possible Solution} to Complex System Modeling}
	\begin{leftanchorblock}{Agent-oriented paradigm}{1}
		Autonomy, distribution, decentralization, proactive, reactive, interaction
	\end{leftanchorblock}
	\only<1>{
		\vspace{-.45cm}
		\begin{columns}
			\begin{column}[t]{.45\linewidth}
				\begin{definitionblock}{Principle of autonomy}
					Agents encapsulate their \Emph{internal state} (that is not accessible to other agents), and make decisions about what to do based on this state, without the direct intervention of humans or others agent
				\end{definitionblock}
				\begin{definitionblock}{Principle of reactive behavior}
					Agent \Emph{perceives} its environment and \Emph{immediately produces an action in response}, without planning or memory
				\end{definitionblock}
			\end{column}
			\begin{column}[t]{.45\linewidth}
				\begin{definitionblock}{Principle of interaction}
					Agent interacts with other agents, and have the ability to engage in social activities (cooperation, coordination, negotation...) in order to achieve its goals
				\end{definitionblock}
				\begin{definitionblock}{Principle of proactive behavior}
					Agent \Emph{pursues its own goals} by anticipating future states and planning actions, independently of immediate stimuli
				\end{definitionblock}
			\end{column}
		\end{columns}
	}
	\only<2->{
		\begin{leftanchorblock}{Holonic multi-agent systems}{2}
			Dynamic hierarchies of agents \cite{CossentinoGaudHilaireGallandKoukam2010_1}
		\end{leftanchorblock}
	}
	\only<2>{
		\begin{columns}
			\begin{column}{.45\linewidth}
				\begin{block}{Complex System \cite{Simon.96}}
					System that has characteristics:
					\begin{itemize}
						\item Hierarchical organization: components are complex systems
						\item Different levels of abstraction
						\item Emergent behavior
						\item Local and nonlinear interactions
					\end{itemize}
				\end{block}
			\end{column}
			\begin{column}{.05\linewidth}
				\centering$\Rightarrow$
			\end{column}
			\begin{column}{.45\linewidth}
				\begin{block}{Holonic Multiagent System \cite{Gerber.99, CossentinoGaudHilaireGallandKoukam2010_1}}
					A multiagent system in which agents may be \Emph{composed of agents} in separated contexts
				\end{block}
			\end{column}
		\end{columns}
	}
	\only<3->{
		\begin{leftanchorblock}{Environment model}{3}
			Infrastructure model, agent perception and action mechanisms \cite{LombardMuallaGallandBuisson2019_1098, BuissonMuallaLombardGalland2019_1100, Michel.07}
		\end{leftanchorblock}
		\begin{leftanchorblock}{Separate Physical component - Decision process}{4}
			Separated models of the cars and of the driving behaviors \cite{saunier2015agent}
		\end{leftanchorblock}
	}
\end{frame}
