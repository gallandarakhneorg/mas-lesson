				Prend en compte les retards implicites dans la commande de contrôle bas niveau
			\end{block}
			\smaller
			\[
			a_r = \dfrac{b_f \tau - 2 v_f \pm 2 b_f \sqrt{\dfrac{b_f b_l \tau^2 + 4 b_l v_f \tau + 4 v_l^2 - 8 b_l s}{4 b_f b_l}}}{2\tau}
			\]
			\smaller
			\begin{itemize}
				\item $s$ : distance entre les deux véhicules successifs
				\item $v_f$, $v_l$ : vitesses actuelles du suiveur et du leader
				\item $b_f$, $b_l$ : décélérations maximales du suiveur et du leader
				\item $\tau$ : temps de réaction du conducteur
			\end{itemize}
