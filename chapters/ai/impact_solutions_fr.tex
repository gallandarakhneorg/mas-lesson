\begin{frame}[t]{Solutions : gouverner, concevoir et éduquer}
	\smaller
	\begin{block}{Cadres réglementaires}
		\begin{compactitemize}
		\item IA Act européen
		\item Moratoires sur certaines applications (reconnaissance faciale en public)
		\item Audits obligatoires d'équité pour les systèmes critiques
		\end{compactitemize}
	\end{block}
	\begin{block}{IA explicable et responsable}
		\begin{compactitemize}
		\item Techniques d'explicabilité (XAI), registres de décisions
		\item Détection de biais a priori, jeux de données de référence diversifiés
		\item Intégration de l'humain dans les systèmes IA
		\end{compactitemize}
	\end{block}
	\begin{block}{Éducation et participation citoyenne}
		\begin{compactitemize}
		\item Former le public aux bases du fonctionnement de l'IA
		\item Inclure des comités d'éthique pluridisciplinaires dès la conception
		\item Soutenir la recherche sur l'IA robuste et alignée avec les valeurs humaines
		\end{compactitemize}
	\end{block}
	\alertbox{L'IA n'est pas une fatalité, c'est un choix collectif}
\end{frame}
