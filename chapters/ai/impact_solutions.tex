\begin{frame}[t]{Solutions: govern, design, and educate}
	\smaller
	\begin{block}{Regulatory frameworks}
		\begin{compactitemize}
		\item European AI Act
		\item Moratoriums on certain applications (facial recognition in public)
		\item Mandatory fairness audits for critical systems
		\end{compactitemize}
	\end{block}
	\begin{block}{Explainable and responsible AI}
		\begin{compactitemize}
		\item Explainability techniques (XAI), decision logs
		\item A priori bias detection, diversified benchmark datasets
		\item Integration of humans in AI systems
		\end{compactitemize}
	\end{block}
	\begin{block}{Education and citizen participation}
		\begin{compactitemize}
		\item Educate the public on the basics of how AI works
		\item Include multidisciplinary ethics committees from the design stage
		\item Support research on robust AI aligned with human values
		\end{compactitemize}
	\end{block}
	\alertbox{AI is not an inevitability, it is a collective choice}
\end{frame}
