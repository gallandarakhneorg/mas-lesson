	\hiconbox{\emph{Outil} conçu par les humains}{public-conclusions-ia-tool}

	\alertbox*{\Emph{À retenir}: L'IA n'est qu'une machine à faire des déductions rapides, pas une intelligence générale}
\end{frame}

\begin{frame}{{Les vrais dangers} ne sont pas "l'IA qui se révolte"}
	\begin{enumerate}
	\item[Biais et discriminations] des données inégalitaires $\rightarrow$ décisions injustes
	\vspace{.25cm}
	\item[Boîte noire] certains modèles restent inexplicables (deep learning, neuro‐symbolique)
	\vspace{.25cm}
	\item[Concentration économique] les grandes entreprises maîtrisant l'IA creusent les écarts
	\end{enumerate}
	\vspace{1cm}
	\begin{exampleblock}{L'Europe agit}
	Règlement sur l'IA, audits d'équité, interdiction de la reconnaissance faciale en public, soutien à l'IA explicable
	\end{exampleblock}
\end{frame}

\begin{frame}{Pourquoi ne pas en avoir peur ?}
	\begin{description}
	\item[On comprend de mieux en mieux] son fonctionnement (recherche en IA explicable)
	\vspace{.25cm}
	\item[On peut la réguler] des lois interdisent les usages les plus dangereux
	\vspace{.25cm}
	\item[On peut la concevoir en intégrant l'humain] comités d'éthique, contrôle humain
	\vspace{.25cm}
	\item[La peur vient souvent de la méconnaissance] cette présentation vous a donné des clés
	\end{description}
	\vspace{1cm}
	\larger
	\alertbox*{L'IA est un outil puissant, ni bon ni mauvais en soi. \\
		C'est à nous, collectivement, de décider de ses usages}
\end{frame}
