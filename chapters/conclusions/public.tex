\begin{frame}{AI is not a magical "super-intelligence"}
	\hiconbox{\emph{Tool} designed by humans}{public-conclusions-ia-tool}
	\hiconbox{Excels at repetitive tasks and large volumes of data (recognizing an image, translating, sorting)}{public-conclusions-ia-algorithm}
	\hiconbox{Does not \emph{think} - calculates probabilities – \Emph{no consciousness, no emotion, no will}}{public-conclusions-ia-nolife}
	\vspace{.5cm}
	\alertbox*{\Emph{Key takeaway}: AI is just a machine that makes fast deductions, not a general intelligence}
\end{frame}

\begin{frame}{{AI is already everywhere...} and it helps us}
	\begin{stabularx}{X|X}
	\tabularhead{In your daily life}{Examples} \\
	Driver assistance systems & Environment perception and reactive behaviors \\
	\hline
	Internet search engines or social networks & Suggestions, spell checking \\
	\hline
	GPS & Route calculation and real-time traffic \\
	\hline
	Email & Anti-spam filters \\
	\hline
	Voice assistants & Speech recognition
	\end{stabularx}
	\vspace{.5cm}
	\alertbox*{AI is above all an \Emph{assistive technology}, freeing up time for humans: creativity, social interaction, complex decisions}
\end{frame}

\begin{frame}{{The real dangers} are not "AI that revolts"}
	\begin{enumerate}
	\item[Bias and discrimination] from unequal data $\rightarrow$ unfair decisions
	\vspace{.25cm}
	\item[Black box] some models remain inexplicable (deep learning, neuro-symbolic)
	\vspace{.25cm}
	\item[Economic concentration] large companies mastering AI widen gaps
	\end{enumerate}
	\vspace{1cm}
	\begin{exampleblock}{Europe takes action}
	AI regulation, fairness audits, ban on facial recognition in public, support for explainable AI
	\end{exampleblock}
\end{frame}

\begin{frame}{Why not be afraid of it?}
	\begin{description}
	\item[We understand it better and better] its functioning (research in explainable AI)
	\vspace{.25cm}
	\item[We can regulate it] laws prohibit the most dangerous uses
	\vspace{.25cm}
	\item[We can design it with human integration] ethics committees, human control
	\vspace{.25cm}
	\item[Fear often comes from lack of knowledge] this presentation has given you the keys
	\end{description}
	\vspace{1cm}
	\larger
	\alertbox*{AI is a powerful tool, neither good nor bad in itself. \\
		It is up to us, collectively, to decide how to use it}
\end{frame}
